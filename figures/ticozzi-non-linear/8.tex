\begin{document}

\begin{tikzpicture}
	\pgfmathsetmacro{\max}{3}
	\draw[->] (-3,0) -- (3,0) node[right] {$x_1$};
	\draw[->] (0,-2) -- (0,2) node[above] {$x_2$};
	
	\draw[domain=-3:3,smooth,variable=\x,blue] plot ({\x},{sqrt(2-\x*\x/(1+\x*\x))});
	\draw[domain=-3:3,smooth,variable=\x,blue] plot ({\x},{-sqrt(2-\x*\x/(1+\x*\x))});
	
	\draw[domain=-3:3,smooth,variable=\x,blue] plot ({\x},{sqrt(1.5-\x*\x/(1+\x*\x))});
	\draw[domain=-3:3,smooth,variable=\x,blue] plot ({\x},{-sqrt(1.5-\x*\x/(1+\x*\x))});
	
	\draw[domain=-0.5:0.5,smooth,variable=\x,blue] plot ({\x},{sqrt(0.2-\x*\x/(1+\x*\x))});
	\draw[domain=-0.5:0.5,smooth,variable=\x,blue] plot ({\x},{-sqrt(0.2-\x*\x/(1+\x*\x))});
	
	\draw[domain=-1:1,smooth,variable=\x,blue] plot ({\x},{sqrt(0.5-\x*\x/(1+\x*\x))});
	\draw[domain=-1:1,smooth,variable=\x,blue] plot ({\x},{-sqrt(0.5-\x*\x/(1+\x*\x))});
	
	\draw[domain=-2:2,smooth,variable=\x,blue] plot ({\x},{sqrt(0.8-\x*\x/(1+\x*\x))});
	\draw[domain=-2:2,smooth,variable=\x,blue] plot ({\x},{-sqrt(0.8-\x*\x/(1+\x*\x))});
\end{tikzpicture}
